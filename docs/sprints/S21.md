class RiskProfile(str, Enum):
    CONSERVATIVE = "conservative"  # Low variance, protect bankroll
    BALANCED = "balanced"          # Moderate risk/reward
    AGGRESSIVE = "aggressive"      # High upside tolerance

class BetStyle(str, Enum):
    PROPS = "props"           # Player prop focus
    SPREADS = "spreads"       # Spread/total focus
    MONEYLINE = "moneyline"   # Straight winners
    PARLAYS = "parlays"       # Multi-leg builder
    SINGLES = "singles"       # One bet at a time

class UserPreferences(BaseModel):
    user_id: UUID
    risk_profile: RiskProfile = RiskProfile.BALANCED
    bet_style: List[BetStyle] = [BetStyle.PARAYS]
    
    # Constraints
    constraints: Constraints = Constraints()
    
    # Bankroll
    bankroll_policy: BankrollPolicy = BankrollPolicy()
    
    # Metadata
    created_at: datetime
    updated_at: datetime
    version: int = 1

class Constraints(BaseModel):
    no_unders: bool = False           # Avoid under bets
    max_legs: int = 6                 # Max parlay legs
    max_correlated_legs: int = 2      # Max legs from same game
    favorite_sports: List[str] = []   # NBA, NFL, etc.
    min_odds: Optional[float] = None  # No bets below X odds
    max_odds: Optional[float] = None  # No bets above X odds
    avoid_teams: List[str] = []       # Never bet on these teams
    avoid_players: List[str] = []     # Never bet on these players

class BankrollPolicy(BaseModel):
    unit_size_percent: float = 1.0    # 1% of bankroll per unit
    max_units_per_bet: float = 3.0    # Max 3 units on any bet
    max_daily_exposure_units: float = 10.0  # Max 10 units/day
    bankroll_goal: Optional[float] = None   # Target growth
    goal_timeline_days: Optional[int] = None  # Days to reach goal
```

def check_constraints(picks: List[Pick], preferences: UserPreferences) -> List[ConstraintViolation]:
    violations = []
    
    # Check max legs
    if len(picks) > preferences.constraints.max_legs:
        violations.append(ConstraintViolation(
            type="max_legs",
            message=f"{len(picks)} legs exceeds your max of {preferences.constraints.max_legs}",
            severity="warning"
        ))
    
    # Check no_unders
    if preferences.constraints.no_unders:
        for pick in picks:
            if "under" in pick.selection.lower():
                violations.append(...)
    
    # Check same-game correlation
    game_counts = Counter(p.game_id for p in picks)
    for game_id, count in game_counts.items():
        if count > preferences.constraints.max_correlated_legs:
            violations.append(...)
    
    return violations
```
